#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <time.h>
#include "lib.h"

enum Durum {
    KIRMIZI_YANIYOR,
    SARI_YANIYOR,
    YESIL_YANIYOR
};

int main() {
    // Rastgele sayı üreticisini başlatıyoruz
    srand(time(NULL)); 

    enum Durum aktif_durum = KIRMIZI_YANIYOR; 
    int kalan_sure = 3; 
    
    int yaya_bekliyor = 0;
    int yaya_sayaci = 0;
    
    printf("Trafik lambasi simulasyonu basliyor...\n");
    printf("Yaya olarak '1' tusuna basabilirsiniz.\n\n");

    while (1) {
        
        // 1. O anki duruma göre ekrana harfi bas
        if (aktif_durum == KIRMIZI_YANIYOR) {
            kirmizi();
        } else if (aktif_durum == SARI_YANIYOR) {
            sari();
        } else if (aktif_durum == YESIL_YANIYOR) {
            yesil();
        }

        // 2. DIŞARIDAN GELEN KESME KONTROLÜ (Tertemiz oldu!)
        if (tusa_basildi_mi() && yaya_bekliyor == 0) {
            yaya_bekliyor = 1;
            yaya_sayaci = (rand() % 10) + 1; // 1 ile 10 arası rastgele
            printf("\n[YAYA BUTONA BASTI! %d saniye icinde zorla kirmizi yanacak]\n\n", yaya_sayaci);
        }

        // 3. Zamanı geçir ve sayaçları düşür
        sleep(1);
        kalan_sure--;
        
        if (yaya_bekliyor == 1) {
            yaya_sayaci--; 
        }

        // 4. YAYA MÜDAHALESİ İLE DURUMU ZORLA DEĞİŞTİRME
        if (yaya_bekliyor == 1 && yaya_sayaci == 0) {
            aktif_durum = KIRMIZI_YANIYOR;  // Sistemi kırmızıya zorla
            kalan_sure = 3;                 // Kırmızı süresini sıfırla
            yaya_bekliyor = 0;              // Yaya bayrağını indir (normale dön)
        } 
        
        // 5. STANDART DÖNGÜ (Senin yazdığın kusursuz değişim mantığı)
        else if (kalan_sure == 0) {
            
            if (aktif_durum == KIRMIZI_YANIYOR) {
                aktif_durum = YESIL_YANIYOR;
                kalan_sure = 5;
            } else if (aktif_durum == SARI_YANIYOR) {
                aktif_durum = KIRMIZI_YANIYOR;
                kalan_sure = 3;
            } else if (aktif_durum == YESIL_YANIYOR) {
                aktif_durum = SARI_YANIYOR;
                kalan_sure = 1;
            }
        }
    }
    
    return 0;
}
